function pickImageModels(allModels) { const imageModels = allModels.filter(m => { const supportsGenerate = (m.supportedGenerationMethods || []).includes('generateContent'); const looksLikeImageModel = /image/i.test(m.name) || /image/i.test(m.displayName || ''); return supportsGenerate && looksLikeImageModel; }); // סדר עדיפות: קודם Nano Banana 2 (3.1 flash), אחר כך 3-pro, אחר כך 2.5, ולבסוף גרסאות lite/preview const priority = [ 'gemini-3.1-flash-image', 'gemini-3-pro-image', 'gemini-2.5-flash-image', 'gemini-3.1-flash-lite-image' ]; imageModels.sort((a, b) => { const nameA = a.name.replace(/^models\//, ''); const nameB = b.name.replace(/^models\//, ''); const aPreview = /preview/i.test(nameA) ? 1 : 0; const bPreview = /preview/i.test(nameB) ? 1 : 0; if (aPreview !== bPreview) return aPreview - bPreview; // יציבים לפני preview const aIdx = priority.indexOf(nameA); const bIdx = priority.indexOf(nameB); const aRank = aIdx === -1 ? priority.length : aIdx; const bRank = bIdx === -1 ? priority.length : bIdx; return aRank - bRank; }); return imageModels.map(m => m.name.replace(/^models\//, '')); }
